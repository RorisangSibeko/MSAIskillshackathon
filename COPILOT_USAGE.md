```python
# Example: Copilot suggested this Azure Maps integration
async def get_safe_route(start_coordinates, end_coordinates, incidents_data):
    """
    Get the safest route between two points using Azure Maps.
    This function was entirely generated by Copilot and then refined.
    """
    # Azure Maps API endpoint
    url = f"https://atlas.microsoft.com/route/directions/json?subscription-key={AZURE_MAPS_KEY}&api-version=1.0&query={start_coordinates[0]},{start_coordinates[1]}:{end_coordinates[0]},{end_coordinates[1]}&routeType=shortest&traffic=true"
    
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as response:
            route_data = await response.json()
            
    # Process route data to avoid incident areas
    route_points = route_data['routes'][0]['legs'][0]['points']
    
    # Check each point against known incidents
    risk_points = []
    for point in route_points:
        for incident in incidents_data:
            distance = calculate_distance(
                (point['latitude'], point['longitude']),
                (incident['latitude'], incident['longitude'])
            )
            if distance < 0.2:  # Within 200 meters of an incident
                risk_points.append({
                    'point': point,
                    'incident': incident,
                    'distance': distance
                })
    
    # If risk points found, request alternative route
    if risk_points:
        # Request alternative route avoiding these areas
        avoid_areas = "&avoid=area:" + ";".join([
            f"{p['point']['latitude']},{p['point']['longitude']},500" 
            for p in risk_points
        ])
        alt_url = url + avoid_areas
        
        async with aiohttp.ClientSession() as session:
            async with session.get(alt_url) as response:
                alt_route_data = await response.json()
        
        return {
            'original_route': route_data,
            'safe_route': alt_route_data,
            'risk_points': risk_points
        }
    
    return {
        'original_route': route_data,
        'safe_route': route_data,
        'risk_points': []
    }
```
